import json
import pandas as pd

# ---- 1. Load ACL JSON ----
with open("acls.json", "r") as f:
    data = json.load(f)

# ---- 2. Flatten into DataFrame ----
rows = []
for acl in data:
    pattern = acl["pattern"]
    entry = acl["entry"]["data"]

    rows.append({
        "user": entry["principal"],
        "topic": pattern["name"],
        "operation": entry["operation"].upper(),
        "permission": entry["permissiontype"].upper()
    })

df = pd.DataFrame(rows)

# ---- 3. Group by user ----
user_ops = df.groupby("user")["operation"].apply(set).reset_index()

# ---- 4. Categorize Users ----
def categorize(ops):
    if "READ" in ops and not any(op in ops for op in ["WRITE", "DELETE", "ALTER"]):
        return "CONSUMER_ONLY"
    elif "WRITE" in ops and not "READ" in ops:
        return "PRODUCER_ONLY"
    elif "READ" in ops and "WRITE" in ops:
        return "HYBRID"
    elif ops == {"DESCRIBE"}:
        return "DESCRIBE_ONLY"
    else:
        return "OTHER"

user_ops["role"] = user_ops["operation"].apply(categorize)

print("\nUser Roles:\n", user_ops)

# ---- 5. Example: List Consumers Only ----
consumers = user_ops[user_ops["role"] == "CONSUMER_ONLY"]
print("\nConsumers Only:\n", consumers)
